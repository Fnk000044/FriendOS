#### 1.2 图标组件重构

**实现内容**:
```typescript
@Component
export struct AppIcon {
  @Prop name: string = '';
  @Prop size: number = 32;
  @Prop variant: 'filled' | 'outline' = 'filled';
  
  build() {
    Stack() {
      // 渐变背景
      Circle()
        .width(this.size)
        .height(this.size)
        .fill(this.getGradient())
      
      // 图标
      Text(this.getIcon())
        .fontSize(this.size * 0.5)
        .fontColor('#FFFFFF')
    }
    .width(this.size)
    .height(this.size)
  }
  
  private getGradient(): ResourceColor {
    // 根据图标名称返回渐变色
    switch (this.name) {
      case 'dashboard': return '#5B9BD5';
      case 'task': return '#06D6A0';
      case 'diary': return '#A8DADC';
      case 'habit': return '#F4A261';
      default: return '#718096';
    }
  }
  
  private getIcon(): string {
    // 返回 emoji 或 unicode 图标
    switch (this.name) {
      case 'dashboard': return '📊';
      case 'task': return '✓';
      case 'diary': return '📝';
      case 'habit': return '🎯';
      default: return '•';
    }
  }
}
```

export class HealthProfileService {

  async generateProfile(userId: string): Promise<HealthProfile> {
    const dimensions = await this.calculateDimensions(userId);
    const insights = await this.generateInsights(dimensions);

```typescript
export class ConversationManager {
  // 对话历史
  private history: Message[] = [];
  
  // 用户画像
  private userProfile: UserProfile | null = null;
  
  // 对话策略
  private strategy: ConversationStrategy | null = null;
  
  // 管理对话
  async manage(userMessage: string): Promise<string> {
    // 1. 更新上下文
    this.updateContext(userMessage);
    
    // 2. 选择策略
    this.selectStrategy();
    
    // 3. 生成响应
    const response = await this.generateResponse(userMessage);
    
    // 4. 更新历史
    this.updateHistory(userMessage, response);
    
    // 5. 更新用户画像
    this.updateUserProfile(userMessage);
    
    return response;
  }
}
```
